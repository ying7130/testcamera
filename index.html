const video = document.getElementById('webcam'); const highResCanvas = document.getElementById('highResCanvas'); const targetCanvas = document.getElementById('targetCanvas'); const highResCtx = highResCanvas.getContext('2d'); const targetCtx = targetCanvas.getContext('2d'); const statusDiv = document.getElementById('status'); const captureBtn = document.getElementById('captureBtn'); const selectionPanel = document.getElementById('selectionPanel'); const optionsContainer = document.getElementById('optionsContainer'); let currentHighResCanvasData = null; // 暫存高畫質 Canvas // 1. 初始化鏡頭 (請求高解析度) async function initCamera() { statusDiv.innerText = "正在啟動鏡頭..."; try { const stream = await navigator.mediaDevices.getUserMedia({ video: { facingMode: { exact: "environment" }, width: { ideal: 1920 }, height: { ideal: 1080 } }, audio: false }); video.srcObject = stream; captureBtn.disabled = false; statusDiv.innerText = "請將品號對準取景框，點擊擷取"; } catch (err) { try { const fallbackStream = await navigator.mediaDevices.getUserMedia({ video: true, audio: false }); video.srcObject = fallbackStream; captureBtn.disabled = false; statusDiv.innerText = "已啟用備用鏡頭"; } catch (e) { statusDiv.innerText = "無法取得鏡頭權限：" + e.message; } } } // 2. 高畫質 AI 辨識 async function captureAndRecognize() { selectionPanel.style.display = "none"; optionsContainer.innerHTML = ""; statusDiv.innerText = "正在高解析度擷取並進行 AI 辨識..."; const vWidth = video.videoWidth || 1280; const vHeight = video.videoHeight || 720; highResCanvas.width = vWidth; highResCanvas.height = vHeight; highResCtx.drawImage(video, 0, 0, vWidth, vHeight); try { const worker = await Tesseract.createWorker('eng'); await worker.setParameters({ // 限寫入英文字母、數字、連字號 -、星號 *、斜線 / tessedit_char_whitelist: 'ABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789-/*', tessedit_pageseg_mode: '6', // 假設圖像為文字區塊 }); const { data: { lines } } = await worker.recognize(highResCanvas); await worker.terminate(); let candidates = []; lines.forEach(line => { // 清理文字：只保留 A-Z, 0-9, -, *, / // 注意：正則中 [-] 放在首位可確保 - 正確被保留 let clean = line.text.replace(/[^A-Za-z0-9\/*-]/g, '').toUpperCase(); // 邏輯過濾： // 1. 長度 >= 4 // 2. 同時包含英文 (A-Z) 與數字 (0-9)，濾除條碼亂碼與價格 const hasLetter = /[A-Z]/.test(clean); const hasNumber = /[0-9]/.test(clean); if (clean.length >= 4 && hasLetter && hasNumber && !candidates.includes(clean)) { candidates.push(clean); } }); if (candidates.length === 0) { statusDiv.innerText = "未辨識出有效品號，請手動輸入："; showSelectionPanel([]); return; } statusDiv.innerText = `辨識成功！請選擇正確品號：`; showSelectionPanel(candidates); } catch (err) { statusDiv.innerText = "辨識發生錯誤：" + err.message; } } // 3. 顯示選單面板 function showSelectionPanel(candidates) { optionsContainer.innerHTML = ""; candidates.forEach(text => { const btn = document.createElement('button'); btn.className = 'option-item'; btn.innerText = text; btn.onclick = () => processAndUpload(text); optionsContainer.appendChild(btn); }); selectionPanel.style.display = "block"; } // 4. 手動輸入提交 function submitManual() { const val = document.getElementById('manualInput').value.trim(); if (!val) { alert("請輸入品號！"); return; } processAndUpload(val); } // 5. 處理轉換與上傳 (關鍵檔名轉換邏輯) function processAndUpload(rawText) { // 再次確保清洗過濾 let cleanText = rawText.replace(/[^A-Za-z0-9\/*-]/g, '').toUpperCase(); // 【檔名判斷規則】 // 只有包含 * 或 / 時才轉全形，- (連字號) 保持半形！ const needsFullWidth = /[*\/]/.test(cleanText); let finalFileName = needsFullWidth ? toFullWidth(cleanText) : cleanText; statusDiv.innerText = `已確認品號：${finalFileName}，正在處理 284x257 圖片...`; // 縮放裁切成 284x257px targetCtx.drawImage(highResCanvas, 0, 0, 284, 257); // 轉為 Base64 並注入 72 DPI const rawTargetDataURL = targetCanvas.toDataURL('image/jpeg', 0.95); const finalImageBase64 = setJpegDpi(rawTargetDataURL, 72); statusDiv.innerText = `上傳完成！檔名：${finalFileName}.jpg (規格：284x257, 72 DPI)`; selectionPanel.style.display = "none"; // 呼叫上傳 uploadToNAS(finalImageBase64, finalFileName); } // 6. 全形轉換（將所有 ASCII 符號/字母轉全形） function toFullWidth(str) { return str.replace(/[\u0021-\u007E]/g, s => String.fromCharCode(s.charCodeAt(0) + 0xFEE0)); } // 7. 寫入 72 DPI Header function setJpegDpi(base64Image, dpi) { const byteString = atob(base64Image.split(',')[1]); const buffer = new Uint8Array(byteString.length); for (let i = 0; i < byteString.length; i++) { buffer[i] = byteString.charCodeAt(i); } if (buffer[0] === 0xFF && buffer[1] === 0xD8) { let offset = 2; while (offset < buffer.length) { if (buffer[offset] === 0xFF && buffer[offset + 1] === 0xE0) { buffer[offset + 11] = 1; // DPI 單位 buffer[offset + 12] = (dpi >> 8) & 0xFF; buffer[offset + 13] = dpi & 0xFF; buffer[offset + 14] = (dpi >> 8) & 0xFF; buffer[offset + 15] = dpi & 0xFF; break; } offset += 1 + buffer[offset + 2]; } } let binary = ''; for (let i = 0; i < buffer.byteLength; i++) { binary += String.fromCharCode(buffer[i]); } return 'data:image/jpeg;base64,' + btoa(binary); } // 8. 上傳至 NAS function uploadToNAS(base64Image, fileName) { console.log("上傳成功！檔名為:", fileName + ".jpg");
